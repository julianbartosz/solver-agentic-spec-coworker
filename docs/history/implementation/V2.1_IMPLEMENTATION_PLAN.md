## 2. GAP-01: Bronze Layer Integration

**Resolves**: DB-003 from Technical Debt Register  
**Priority**: P3 (Low - audit compliance, not functional)  
**Estimated Effort**: 4 hours

### 2.1 Current State

The `spec_bronze` schema exists in Postgres (per `postgres.py`), but `ingest_spec.py` does **not** store raw spec bytes:

```python
# Current ingest_spec.py - no bronze layer writes
def ingest_spec(state: WorkflowState) -> WorkflowState:
    # Fetches spec content
    # Stores in state.spec_documents (Silver)
    # Does NOT persist raw bytes to spec_bronze.raw_specs
```

### 2.2 Target Architecture

Store raw spec bytes on ingest for audit replay:

```
ingest_spec() 
  → fetch raw bytes
  → store in spec_bronze.raw_specs (sha256 dedup)
  → continue to Silver processing
```

### 2.3 Implementation

**File: `src/integration_coworker/persistence/bronze.py`** (NEW)

```python
"""
Bronze layer persistence for raw spec storage.

Stores unprocessed spec content for audit trail and replay capability.
Per DB-003 in Technical Debt Register.
"""
import hashlib
import logging
from typing import Optional, Tuple
from datetime import datetime

from integration_coworker.persistence.db import get_connection, get_engine_type

logger = logging.getLogger(__name__)


def store_raw_spec(
    source_system_id: int,
    uri: str,
    raw_content: bytes,
    content_type: str,
) -> Tuple[int, bool]:
    """
    Store raw spec content in Bronze layer.
    
    Uses SHA256 deduplication - if identical content exists, returns existing ID.
    
    Args:
        source_system_id: FK to source_systems table
        uri: Original spec URI
        raw_content: Raw bytes of spec file
        content_type: MIME type (e.g., "application/json", "text/yaml")
        
    Returns:
        Tuple of (raw_spec_id, was_new)
        was_new is False if content already existed (dedup hit)
    """
    sha256 = hashlib.sha256(raw_content).hexdigest()
    
    engine = get_engine_type()
    
    if engine == "postgres":
        from integration_coworker.persistence.postgres import get_connection as pg_conn
        
        with pg_conn() as conn:
            with conn.cursor() as cur:
                # Check if already exists (dedup)
                cur.execute("""
                    SELECT id FROM spec_bronze.raw_specs
                    WHERE source_system_id = %s AND sha256 = %s
                """, (source_system_id, sha256))
                
                existing = cur.fetchone()
                if existing:
                    logger.debug(f"Bronze dedup hit for {uri} (sha256: {sha256[:12]}...)")
                    return (existing[0], False)
                
                # Insert new
                cur.execute("""
                    INSERT INTO spec_bronze.raw_specs 
                        (source_system_id, uri, raw_content, content_type, sha256)
                    VALUES (%s, %s, %s, %s, %s)
                    RETURNING id
                """, (source_system_id, uri, raw_content, content_type, sha256))
                
                new_id = cur.fetchone()[0]
                conn.commit()
                logger.info(f"Stored raw spec in Bronze: {uri} -> id={new_id}")
                return (new_id, True)
    
    else:
        # SQLite fallback
        conn = get_connection()
        cur = conn.cursor()
        
        cur.execute("""
            SELECT id FROM raw_specs
            WHERE source_system_id = ? AND sha256 = ?
        """, (source_system_id, sha256))
        
        existing = cur.fetchone()
        if existing:
            conn.close()
            return (existing[0], False)
        
        cur.execute("""
            INSERT INTO raw_specs (source_system_id, uri, raw_content, content_type, sha256)
            VALUES (?, ?, ?, ?, ?)
        """, (source_system_id, uri, raw_content, content_type, sha256))
        
        new_id = cur.lastrowid
        conn.commit()
        conn.close()
        return (new_id, True)


def get_raw_spec(raw_spec_id: int) -> Optional[bytes]:
    """
    Retrieve raw spec content by ID.
    
    Used for audit replay.
    """
    engine = get_engine_type()
    
    if engine == "postgres":
        from integration_coworker.persistence.postgres import get_connection as pg_conn
        
        with pg_conn() as conn:
            with conn.cursor() as cur:
                cur.execute("""
                    SELECT raw_content FROM spec_bronze.raw_specs WHERE id = %s
                """, (raw_spec_id,))
                row = cur.fetchone()
                return row[0] if row else None
    else:
        conn = get_connection()
        cur = conn.cursor()
        cur.execute("SELECT raw_content FROM raw_specs WHERE id = ?", (raw_spec_id,))
        row = cur.fetchone()
        conn.close()
        return row[0] if row else None
```

```python
# Add new function for runtime-style client generation

def _build_runtime_client_code(
    state: WorkflowState,
    endpoint: Endpoint,
    client_class: str,
    method_name: str,
) -> str:
    """
    Generate thin client code using runtime library.
    
    V2.1: Runtime-first codegen per ADR-0005.
    Produces ~30 LOC instead of ~300 LOC inline clients.
    """
    provider = state.provider_code or "api"
    base_url = _extract_base_url(state.spec_documents[0] if state.spec_documents else None)
    env_var = f"{provider.upper()}_API_KEY"
    
    # Determine auth type from policies
    auth_class = "NoAuth"
    auth_args = ""
    for policy in state.policies:
        if policy.policy_type == "auth":
            if policy.implementation_hint == "bearer":
                auth_class = "BearerAuth"
                auth_args = f'env_var="{env_var}"'
            elif policy.implementation_hint == "api_key":
                auth_class = "ApiKeyAuth"
                header = policy.config.get("header", "X-API-Key")
                auth_args = f'env_var="{env_var}", header="{header}"'
    
    # Determine retry from policies
    retry_class = "NoRetry"
    retry_args = ""
    for policy in state.policies:
        if policy.policy_type == "retry":
            retry_class = "ExponentialRetry"
            max_attempts = policy.config.get("max_attempts", 3)
            retry_args = f"max_attempts={max_attempts}"
    
    # Determine rate limit from policies
    rate_limit_class = "NoRateLimiter"
    rate_limit_args = ""
    for policy in state.policies:
        if policy.policy_type == "rate_limit":
            rate_limit_class = "TokenBucketRateLimiter"
            rps = policy.config.get("rps", 10)
            rate_limit_args = f"rps={rps}"
    
    # Build method signature
    params = _build_method_params(endpoint)
    http_method = (endpoint.method or "GET").lower()
    path = endpoint.path or "/"
    
    # Build request body if needed
    body_arg = ""
    if http_method in ("post", "put", "patch"):
        body_arg = ", json=request_body"
    
    return f'''"""
Auto-generated {provider.title()} API client using runtime library.

Generated by integration-coworker v2.1
"""
from integration_coworker.runtime import (
    IntegrationClient,
    {auth_class},
    {retry_class},
    {rate_limit_class},
)


class {client_class}(IntegrationClient):
    """Client for {provider.title()} API."""
    
    def __init__(self):
        super().__init__(
            base_url="{base_url}",
            auth={auth_class}({auth_args}),
            retry={retry_class}({retry_args}),
            rate_limiter={rate_limit_class}({rate_limit_args}),
        )
    
    def {method_name}(self{params}) -> dict:
        """
        {endpoint.summary or f'{http_method.upper()} {path}'}
        """
        return self.{http_method}("{path}"{body_arg})
'''


# Update _generate_client_code to use policy_mode:

def _generate_client_code(state: WorkflowState, ...) -> str:
    """Generate client code with configurable policy mode."""
    
    # V2.1: Check policy_mode (GAP-02)
    options = getattr(state, 'options', None)
    policy_mode = getattr(options, 'policy_mode', 'inline') if options else 'inline'
    
    if policy_mode == "runtime":
        return _build_runtime_client_code(state, endpoint, client_class, method_name)
    else:
        # Existing inline generation
        return _build_inline_client_code(state, endpoint, client_class, method_name)
```

```python
                cur.execute("SELECT 1")

        """Verify all required schemas exist."""
        from integration_coworker.persistence.postgres import get_connection
        
        with get_connection() as conn:
            with conn.cursor() as cur:
                cur.execute("""
                    SELECT schema_name FROM information_schema.schemata
                    WHERE schema_name IN ('spec_bronze', 'spec_silver', 'integration_gold', 'kg')
                """)
                schemas = {row[0] for row in cur.fetchall()}
        
        expected = {'spec_bronze', 'spec_silver', 'integration_gold', 'kg'}

    def test_pgvector_extension(self):

        with get_connection() as conn:
            with conn.cursor() as cur:
```

```python
def _compute_graph_score(
    template: KGWorkflowTemplate,
    entities: List[str],
    provider_code: Optional[str] = None,
) -> float:
    """
    Compute graph-based relevance score using actual KG edges.
    
    V2.1: Queries kg_edges table for real relationship data.
    
    Scoring factors:
    - Provider match: +0.3 if template provider matches target
    - Edge density: Higher score for templates with more edges
    - Entity coverage: Bonus for edges to known entities
    
    Args:
        template: KGWorkflowTemplate with node_id
        entities: Entity names from task understanding
        provider_code: Current provider for match bonus
        
    Returns:
        Score in [0, 1] range
    """
    # Check if we have a node_id to query
    node_id = getattr(template, 'node_id', None)
    if not node_id:
        # Fall back to text matching if no node_id
        return _compute_graph_score_fallback(template, entities, provider_code)
    
    try:
        from integration_coworker.persistence.db import get_engine_type
        
        engine = get_engine_type()
        
        # Edge type weights
        EDGE_WEIGHTS = {
            "uses_endpoint": 1.0,
            "references_entity": 0.7,
            "has_step": 0.5,
            "belongs_to_provider": 0.3,
            "implements_pattern": 0.4,
        }
        
        if engine == "postgres":
            from integration_coworker.persistence.postgres import get_connection
            
            with get_connection() as conn:
                with conn.cursor() as cur:
                    # Count edges by type from this template node
                    cur.execute("""
                        SELECT relation_type, COUNT(*) as cnt
                        FROM kg.edges
                        WHERE src_node_id = %s OR dst_node_id = %s
                        GROUP BY relation_type
                    """, (node_id, node_id))
                    
                    edge_counts = {row[0]: row[1] for row in cur.fetchall()}
                    
                    # Count edges to entity nodes (if we have entities)
                    entity_edge_count = 0
                    if entities:
                        # Build safe placeholder list
                        placeholders = ','.join(['%s'] * len(entities))
                        cur.execute(f"""
                            SELECT COUNT(DISTINCT e.id)
                            FROM kg.edges e
                            JOIN kg.nodes n ON (e.dst_node_id = n.id OR e.src_node_id = n.id)
                            WHERE (e.src_node_id = %s OR e.dst_node_id = %s)
                            AND n.node_key IN ({placeholders})
                        """, [node_id, node_id] + [e.lower() for e in entities])
                        entity_edge_count = cur.fetchone()[0] or 0
        else:
            # SQLite implementation
            from integration_coworker.persistence.db import get_connection
            conn = get_connection()
            cur = conn.cursor()
            
            cur.execute("""
                SELECT relation_type, COUNT(*) as cnt
                FROM kg_edges
                WHERE src_node_id = ? OR dst_node_id = ?
                GROUP BY relation_type
            """, (node_id, node_id))
            
            edge_counts = {row[0]: row[1] for row in cur.fetchall()}
            entity_edge_count = 0  # Simplified for SQLite
            conn.close()
        
        # Calculate weighted score
        total_weight = 0.0
        for edge_type, count in edge_counts.items():
            weight = EDGE_WEIGHTS.get(edge_type, 0.1)
            total_weight += min(count, 5) * weight  # Cap at 5 edges per type
        
        # Add entity relevance bonus
        if entities and entity_edge_count > 0:
            entity_bonus = min(1.0, entity_edge_count / len(entities))
            total_weight += entity_bonus * 2.0
        
        # Normalize to [0, 1] - max reasonable score ~10
        normalized_score = min(1.0, total_weight / 10.0)
        
        # Provider match bonus
        template_provider = getattr(template, 'provider_code', None)
        if provider_code and template_provider == provider_code:
            normalized_score = min(1.0, normalized_score + 0.15)
        
        return normalized_score
        
    except Exception as e:
        logger.warning(f"KG edge query failed, falling back to text match: {e}")
        return _compute_graph_score_fallback(template, entities, provider_code)


def _compute_graph_score_fallback(
    template: KGWorkflowTemplate,
    entities: List[str],
    provider_code: Optional[str] = None,
) -> float:
    """
    Fallback graph score using text matching.
    
    Used when KG edge query fails or template has no node_id.
    """
    if not entities:
        return 0.3 if provider_code else 0.1
    
    template_text = (
        f"{getattr(template, 'name', '')} {getattr(template, 'description', '')} "
        f"{' '.join(s.get('label', '') for s in getattr(template, 'steps', []))}"
    ).lower()
    
    matches = sum(1 for e in entities if e.lower() in template_text)
    return min(1.0, matches / len(entities))
```

## Fix #1: Constrained Path Generation (Path Hallucination)

### Problem
LLM hallucinates non-existent API paths (`/AlphaSenders`, `/Compliance/Usa2p/`) even when given explicit valid path lists.

### Solution
Inject paths from spec into code template BEFORE LLM refinement. LLM only fills method body logic.

def build_constrained_body_prompt(
    template_code: str,
    endpoint: 'Endpoint',
    path_params: List[str],
    query_params: List[str],
    has_body: bool,
) -> str:
    """
    Build prompt for filling ONLY the method body in constrained generation.
    
    The LLM receives the template and fills the IMPLEMENTATION_PLACEHOLDER.
    It cannot modify paths, methods, or class structure.
    
    V2.3 Fix for path hallucination.
    """

    if path_params:
        param_docs.append(f"Path parameters: {', '.join(path_params)}")
    if query_params:
        param_docs.append(f"Query parameters: {', '.join(query_params)}")
    if has_body:
        param_docs.append("Request body: payload (Dict[str, Any])")
    
    params_section = "\n".join(param_docs) if param_docs else "No parameters"
    
    return f"""
Fill in ONLY the method body where you see "# IMPLEMENTATION_PLACEHOLDER".

CRITICAL RULES:
1. Do NOT change the endpoint path (it's fixed from the spec)
2. Do NOT change the HTTP method (it's fixed from the spec)
3. Do NOT change the class name or method signature
4. ONLY implement the logic inside the method body

Template:
```python
{template_code}
```

Method parameters:
{params_section}

Implementation requirements:
1. Build the URL with path parameters (use f-string substitution)
2. Add query parameters to the request if provided
3. Make the HTTP request using self._client
4. Handle errors and wrap in IntegrationError
5. Return the JSON response as a dict

Generate ONLY the method body code (the lines that replace "pass").
Do NOT include the full class or method signature.

"""
Unified Semantic Index for all vector search operations.

V2.3: Single interface for searching spec chunks, KG templates, and run history.
"""
import logging
from dataclasses import dataclass
from typing import List, Optional, Literal

from integration_coworker.retrieval.semantic_search import (
    search_spec_chunks,
    embed_text,
)
from integration_coworker.kg import search_kg_templates

logger = logging.getLogger(__name__)


@dataclass
class SearchResult:
    """Unified search result from any index."""
    content: str
    score: float
    source: Literal["spec_chunk", "kg_template", "run_history"]
    metadata: dict


class UnifiedSemanticIndex:
    """
    Single interface for all semantic search in the system.
    
    Queries:
    1. Spec chunks (what does the API do?)
    2. KG templates (what patterns exist?)
    3. Previous runs (what worked before?) - future
    """
    
    def search(
        self,
        query: str,
        index_types: Optional[List[str]] = None,
        top_k: int = 10,
        provider_code: Optional[str] = None,
        spec_document_id: Optional[int] = None,
    ) -> List[SearchResult]:
        """
        Unified semantic search across all indices.
        
        Args:
            query: Search query text
            index_types: List of indices to search. Default: ["spec_chunks", "kg_templates"]
            top_k: Number of results per index
            provider_code: Filter by provider
            spec_document_id: Filter by spec document
            
        Returns:
            Combined and re-ranked results from all indices
        """
        if index_types is None:
            index_types = ["spec_chunks", "kg_templates"]
        
        results = []
        
        if "spec_chunks" in index_types:
            try:
                chunks = search_spec_chunks(
                    query,
                    top_k=top_k,
                    spec_document_id=spec_document_id,
                )
                for chunk in chunks:
                    results.append(SearchResult(
                        content=chunk.content if hasattr(chunk, 'content') else str(chunk),
                        score=chunk.score if hasattr(chunk, 'score') else 0.5,
                        source="spec_chunk",
                        metadata={"chunk_id": getattr(chunk, 'id', None)},
                    ))
            except Exception as e:
                logger.debug(f"Spec chunk search failed: {e}")
        
        if "kg_templates" in index_types:
            try:
                templates = search_kg_templates(
                    query,
                    provider_code=provider_code,
                    top_k=top_k,
                )
                for template in templates:
                    results.append(SearchResult(
                        content=template.description if hasattr(template, 'description') else str(template),
                        score=template.score if hasattr(template, 'score') else 0.5,
                        source="kg_template",
                        metadata={"template_key": getattr(template, 'key', None)},
                    ))
            except Exception as e:
                logger.debug(f"KG template search failed: {e}")
        
        # Sort by score descending and limit
        results.sort(key=lambda r: r.score, reverse=True)
        return results[:top_k]
    
    def search_for_context(
        self,
        query: str,
        provider_code: Optional[str] = None,
        max_tokens: int = 2000,
    ) -> str:
        """
        Search and format results as context for LLM prompts.
        
        Args:
            query: Search query
            provider_code: Filter by provider
            max_tokens: Approximate token limit for context
            
        Returns:
            Formatted context string for LLM
        """
        results = self.search(query, provider_code=provider_code, top_k=10)
        
        context_parts = []
        total_chars = 0
        char_limit = max_tokens * 4  # Rough chars per token
        
        for result in results:
            if total_chars + len(result.content) > char_limit:
                break
            context_parts.append(f"[{result.source}] {result.content[:500]}")
            total_chars += len(result.content[:500])
        
        return "\n---\n".join(context_parts)


# Module-level singleton for convenience
_index: Optional[UnifiedSemanticIndex] = None


def get_unified_index() -> UnifiedSemanticIndex:
    """Get or create the unified semantic index singleton."""
    global _index
    if _index is None:
        _index = UnifiedSemanticIndex()
    return _index
```

def _augment_policies_with_llm(
    spec_policies: List[Policy],
    state: WorkflowState,
) -> List[Policy]:
    """
    Augment spec-extracted policies with LLM analysis of prose.
    
    V2.3: Hybrid policy inference.
    
    Args:
        spec_policies: Policies already extracted from OpenAPI schema
        state: WorkflowState with spec sections
        
    Returns:
        Combined policies (spec + LLM-discovered)
    """
    # Only augment if we have prose sections
    if not state.spec_sections:
        return spec_policies
    
    # Filter for sections likely to contain policy info
    prose_sections = [
        s for s in state.spec_sections 
        if hasattr(s, 'section_type') and s.section_type in (
            "description", "overview", "rate_limits", "authentication", "errors"
        )
    ]
    
    if not prose_sections:
        return spec_policies
    
    try:
        from integration_coworker.llm import call_llm_json
        
        # Format existing policies
        existing = [
            {"type": p.policy_type.value if hasattr(p.policy_type, 'value') else str(p.policy_type)}
            for p in spec_policies
        ]
        
        # Format prose sections (limit to avoid token overflow)
        prose_text = "\n\n".join([
            f"## {s.title if hasattr(s, 'title') else 'Section'}\n{s.content[:1000] if hasattr(s, 'content') else ''}"
            for s in prose_sections[:5]
        ])
        
        prompt = f"""
Analyze this API documentation for policies NOT already captured:

{prose_text}

Already detected from OpenAPI schema:
{existing}

Find ADDITIONAL policies mentioned in the documentation. Look for:
1. Rate limits (e.g., "100 requests per minute", "X-RateLimit headers")
2. Retry guidance (e.g., "retry after 60 seconds", "exponential backoff")
3. Pagination (e.g., "use cursor", "max 100 per page")
4. Idempotency (e.g., "Idempotency-Key header")

Return JSON array of new policies only (not duplicates):
[
  {{"type": "rate_limit", "config": {{"rps": 100}}, "source": "Found in: ..."}}
]

If no additional policies found, return empty array: []
"""
        
        result = call_llm_json(prompt)
        
        if result and isinstance(result, list):
            for item in result:
                if isinstance(item, dict) and "type" in item:
                    policy_type = PolicyType(item["type"]) if item["type"] in [e.value for e in PolicyType] else None
                    if policy_type:
                        spec_policies.append(Policy(
                            id=None,
                            task_id=None,
                            policy_type=policy_type,
                            config=item.get("config", {}),
                        ))
                        logger.info(f"LLM discovered policy: {item['type']} - {item.get('source', 'prose')}")
        
        return spec_policies
        
    except Exception as e:
        logger.debug(f"LLM policy augmentation failed: {e}")
        return spec_policies
```

"""
Async LLM Request Batcher for concurrent execution.

V2.3: Batches LLM requests across concurrent runs to respect rate limits.
"""
import asyncio
import logging
import time
from dataclasses import dataclass
from typing import Optional, Callable, Any
from concurrent.futures import Future
import threading

logger = logging.getLogger(__name__)


@dataclass
class LLMRequest:
    """A queued LLM request."""
    prompt: str
    run_id: str
    future: Future
    submitted_at: float


class AsyncLLMBatcher:
    """
    Batches LLM requests across concurrent runs for efficiency.
    
    Instead of each run making sequential calls, this queues requests
    and processes them with controlled concurrency.
    """
    
    def __init__(
        self,
        llm_fn: Callable[[str], str],
        max_concurrent: int = 3,
        batch_delay_ms: int = 100,
    ):
        """
        Initialize the batcher.
        
        Args:
            llm_fn: The LLM call function (prompt -> response)
            max_concurrent: Max parallel LLM calls
            batch_delay_ms: How long to wait for batch to fill
        """
        self._llm_fn = llm_fn
        self._max_concurrent = max_concurrent
        self._batch_delay_ms = batch_delay_ms
        self._queue: list = []
        self._lock = threading.Lock()
        self._semaphore = threading.Semaphore(max_concurrent)
        self._running = False
        self._worker_thread: Optional[threading.Thread] = None
    
    def start(self):
        """Start the background worker."""
        if self._running:
            return
        self._running = True
        self._worker_thread = threading.Thread(target=self._process_loop, daemon=True)
        self._worker_thread.start()
        logger.debug(f"AsyncLLMBatcher started (max_concurrent={self._max_concurrent})")
    
    def stop(self):
        """Stop the background worker."""
        self._running = False
        if self._worker_thread:
            self._worker_thread.join(timeout=1.0)
    
    def submit(self, prompt: str, run_id: str = "unknown") -> Future:
        """
        Submit an LLM request and return a Future.
        
        Args:
            prompt: The LLM prompt
            run_id: Run identifier for tracking
            
        Returns:
            Future that will contain the response
        """
        future = Future()
        request = LLMRequest(
            prompt=prompt,
            run_id=run_id,
            future=future,
            submitted_at=time.time(),
        )
        
        with self._lock:
            self._queue.append(request)
        
        return future
    
    def _process_loop(self):
        """Background loop processing requests."""
        while self._running:
            # Collect batch
            time.sleep(self._batch_delay_ms / 1000)
            
            with self._lock:
                if not self._queue:
                    continue
                batch = self._queue[:self._max_concurrent]
                self._queue = self._queue[self._max_concurrent:]
            
            # Process batch in parallel threads
            threads = []
            for request in batch:
                t = threading.Thread(target=self._process_request, args=(request,))
                t.start()
                threads.append(t)
            
            for t in threads:
                t.join()
    
    def _process_request(self, request: LLMRequest):
        """Process a single request with semaphore control."""
        with self._semaphore:
            try:
                response = self._llm_fn(request.prompt)
                request.future.set_result(response)
                logger.debug(f"Completed LLM request for run {request.run_id}")
            except Exception as e:
                request.future.set_exception(e)
                logger.warning(f"LLM request failed for run {request.run_id}: {e}")


# Module-level singleton
_batcher: Optional[AsyncLLMBatcher] = None


def get_llm_batcher(llm_fn: Optional[Callable] = None) -> AsyncLLMBatcher:
    """Get or create the LLM batcher singleton."""
    global _batcher
    if _batcher is None:
        if llm_fn is None:
            from integration_coworker.llm.client import call_llm
            llm_fn = call_llm
        _batcher = AsyncLLMBatcher(llm_fn)
        _batcher.start()
    return _batcher
